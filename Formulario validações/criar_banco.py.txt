import pandas as pd
import sqlite3


# =========================
# ARQUIVO EXCEL
# =========================
arquivo = "Produtos_modulacoes_colecao (2).xlsx"


# =========================
# LER EXCEL
# =========================
excel = pd.read_excel(arquivo)


# =========================
# MOSTRAR COLUNAS
# =========================
print(excel.columns)


# =========================
# AJUSTAR NOMES
# =========================
excel.columns = [
    c.strip().lower()
    for c in excel.columns
]


# =========================
# CONECTAR SQLITE
# =========================
conn = sqlite3.connect("produtos.db")


# =========================
# SALVAR TABELA SQL
# =========================
excel.to_sql(
    "produtos",
    conn,
    if_exists="replace",
    index=False
)


# =========================
# FECHAR CONEXAO
# =========================
conn.close()


# =========================
# FINAL
# =========================
print("Banco criado com sucesso!")
